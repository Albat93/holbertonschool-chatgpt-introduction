import tkinter as tk from tkinter import messagebox class Checkbook: def __init__(self): self.balance = 0.0 def deposit(self, amount): if amount <= 0: messagebox.showerror("Error", "Invalid deposit amount. Please enter a positive number.") return self.balance += amount messagebox.showinfo("Deposit", f"Deposited ${amount:.2f}\nCurrent Balance: ${self.balance:.2f}") def withdraw(self, amount): if amount <= 0: messagebox.showerror("Error", "Invalid withdrawal amount. Please enter a positive number.") return if amount > self.balance: messagebox.showerror("Error", "Insufficient funds to complete the withdrawal.") else: self.balance -= amount messagebox.showinfo("Withdrawal", f"Withdrew ${amount:.2f}\nCurrent Balance: ${self.balance:.2f}") def get_balance(self): messagebox.showinfo("Balance", f"Current Balance: ${self.balance:.2f}") class Application(tk.Tk): def __init__(self): super().__init__() self.cb = Checkbook() self.title("Checkbook Application") self.geometry("400x300") self.config(bg="lightgray") # Create buttons self.deposit_button = tk.Button(self, text="Deposit", command=self.deposit_action) self.deposit_button.pack(pady=10) self.withdraw_button = tk.Button(self, text="Withdraw", command=self.withdraw_action) self.withdraw_button.pack(pady=10) self.balance_button = tk.Button(self, text="Check Balance", command=self.check_balance_action) self.balance_button.pack(pady=10) self.exit_button = tk.Button(self, text="Exit", command=self.quit) self.exit_button.pack(pady=10) def deposit_action(self): amount = self.get_amount("deposit") if amount is not None: self.cb.deposit(amount) self.config(bg="lightgreen") # Change background to green on deposit def withdraw_action(self): amount = self.get_amount("withdraw") if amount is not None: self.cb.withdraw(amount) self.config(bg="lightcoral") # Change background to red on withdrawal def check_balance_action(self): self.cb.get_balance() self.config(bg="lightblue") # Change background to blue when checking balance def get_amount(self, action): try: amount = float(tk.simpledialog.askstring("Input", f"Enter the amount to {action}: $")) if amount <= 0: messagebox.showerror("Error", f"Invalid {action} amount. Please enter a positive number.") return None return amount except (ValueError, TypeError): messagebox.showerror("Error", "Invalid input. Please enter a numeric value.") return None if __name__ == "__main__": app = Application() app.mainloop()
